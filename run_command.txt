# ── BƯỚC 1: Build VECTOR cache (KHUYẾN NGHỊ — chỉ ~1MB RAM) ─────
python scripts/build_vector_cache.py \
  --train_csv data/fer13-split/train.csv \
  --val_csv   data/fer13-split/val.csv \
  --test_csv  data/fer13-split/test.csv \
  --save_dir  outputs/vector_cache

# ── BƯỚC 2: Train MLP baseline ────────────────────────────────────
python -m scripts.train --config mlp_baseline --env local
python -m scripts.train --config mlp_baseline --env kaggle

# ── (Tuỳ chọn) Build FULL graph cache (GCN tương lai — ~24GB RAM!) ──
python scripts/build_graph_cache.py \
  --train_csv data/fer13-split/train.csv \
  --val_csv   data/fer13-split/val.csv \
  --test_csv  data/fer13-split/test.csv \
  --save_dir  outputs/graph_cache

# ── (Tuỳ chọn) Kiểm tra graph cache ─────────────────────────────
python scripts/check_graph_cache.py \
  --graph_path outputs/graph_cache/train_graphs.pt \
  --index 0

# ── Check toàn bộ pipeline ────────────────────────────────────────
python scripts/check_pipeline.py
